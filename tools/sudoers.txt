# RPi Admin UI - allowlisted systemctl/journalctl commands for user pi
# Paths confirmed:
# /usr/bin/systemctl
# /usr/bin/journalctl

# --- rpi-admin-ui.service ---
pi ALL=(root) NOPASSWD: /usr/bin/systemctl start rpi-admin-ui.service
pi ALL=(root) NOPASSWD: /usr/bin/systemctl stop rpi-admin-ui.service
pi ALL=(root) NOPASSWD: /usr/bin/systemctl restart rpi-admin-ui.service
pi ALL=(root) NOPASSWD: /usr/bin/systemctl is-active rpi-admin-ui.service
pi ALL=(root) NOPASSWD: /usr/bin/systemctl status rpi-admin-ui.service
pi ALL=(root) NOPASSWD: /usr/bin/journalctl -u rpi-admin-ui.service *
